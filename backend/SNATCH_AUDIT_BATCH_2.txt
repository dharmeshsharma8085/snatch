load_dotenv()

JWT_SECRET_KEY = os.getenv("JWT_SECRET_KEY")
JWT_ALGORITHM = os.getenv("JWT_ALGORITHM", "HS256")
ACCESS_TOKEN_EXPIRE_MINUTES = int(
    os.getenv("ACCESS_TOKEN_EXPIRE_MINUTES", "60")
)

if not JWT_SECRET_KEY or len(JWT_SECRET_KEY) < 32:
    raise RuntimeError(
        "JWT_SECRET_KEY must contain at least 32 characters"
    )

password_hasher = PasswordHash.recommended()


def hash_password(password: str) -> str:
    return password_hasher.hash(password)


def verify_password(password: str, password_hash: str) -> bool:
    return password_hasher.verify(password, password_hash)


def create_access_token(user_id: str) -> str:
    now = datetime.now(timezone.utc)
    expires_at = now + timedelta(
        minutes=ACCESS_TOKEN_EXPIRE_MINUTES
    )

    payload = {
        "sub": user_id,
        "exp": expires_at,
        "iat": now,
    }

    return jwt.encode(
        payload,
        JWT_SECRET_KEY,
        algorithm=JWT_ALGORITHM,
    )


def register_user(data: UserRegister) -> UserResponse:
    email = str(data.email).strip().lower()

    with SessionLocal() as db:
        existing_user = db.scalar(
            select(User).where(User.email == email)
        )

        if existing_user:
            raise ValueError("Email is already registered")

        user = User(
            email=email,
            password_hash=hash_password(data.password),
            is_active=True,
        )

        try:
            db.add(user)
            db.commit()
            db.refresh(user)

            return UserResponse(
                user_id=user.user_id,
                email=user.email,
                is_active=user.is_active,
            )

        except IntegrityError:
            db.rollback()
            raise ValueError("Email is already registered")


def login_user(data: UserLogin) -> TokenResponse:
    email = str(data.email).strip().lower()

    with SessionLocal() as db:
        user = db.scalar(
            select(User).where(User.email == email)
        )

        if not user or not verify_password(
            data.password, user.password_hash
        ):
            raise ValueError("Invalid email or password")

        if not user.is_active:
            raise ValueError("Account is inactive")

        token = create_access_token(user.user_id)

        return TokenResponse(
            access_token=token,
            token_type="bearer",
            expires_in=ACCESS_TOKEN_EXPIRE_MINUTES * 60,
        )

import logging

from vonage import Auth, Vonage
from vonage_sms import SmsMessage

from app.services.sms_provider import SMSResult

logger = logging.getLogger(__name__)


class VonageSMSProvider:
    def __init__(
        self,
        api_key: str,
        api_secret: str,
        sender_id: str,
    ):
        self.client = Vonage(
            Auth(
                api_key=api_key,
                api_secret=api_secret,
            )
        )
        self.sender_id = sender_id
    def build_message(
        self,
        phone_number: str,
        message: str,
    ) -> SmsMessage:
        if not phone_number.strip():
            raise ValueError("Phone number cannot be empty.")

        if not message.strip():
            raise ValueError("Message cannot be empty.")

        return SmsMessage(
            to=phone_number,
            from_=self.sender_id,
            text=message,
        )
    
    

    def send_sms(
        self,
        phone_number: str,
        message: str,
    ) -> SMSResult:

        # Validation happens before contacting Vonage.
        try:
            sms = self.build_message(phone_number, message)
        except ValueError as error:
            return SMSResult(
                success=False,
                error=str(error),
            )

        # Once the network request starts, an exception
        # may mean Vonage accepted the SMS but the response
        # was lost. Treat it as an unknown outcome.
        try:
            response = self.client.sms.send(sms)
        except Exception as error:
            logger.error(
                "Vonage request outcome unknown. Exception type: %s",
                type(error).__name__,
            )
            return SMSResult(
                success=False,
                error="Vonage request outcome is unknown. Manual reconciliation required.",
                outcome_unknown=True,
            )

        try:
            response_data = response.model_dump(exclude_unset=True)
            messages = response_data.get("messages", [])

            if not messages:
                return SMSResult(
                    success=False,
                    error="Vonage returned an empty response. Outcome unknown.",
                    outcome_unknown=True,
                )

            result = messages[0]

            if str(result.get("status")) == "0":
                message_id = result.get("message_id")

                if not message_id:
                    return SMSResult(
                        success=False,
                        error="Vonage accepted the request without a message ID.",
                        outcome_unknown=True,
                    )

                return SMSResult(
                    success=True,
                    provider_message_id=message_id,
                )

            return SMSResult(
                success=False,
                error=result.get(
                    "error_text",
                    "Vonage rejected the SMS request.",
                ),
            )

        except Exception as error:
            logger.error(
                "Vonage response could not be processed. Exception type: %s",
                type(error).__name__,
            )
            return SMSResult(
                success=False,
                error="Vonage response could not be verified. Outcome unknown.",
                outcome_unknown=True,
            )

from dataclasses import dataclass
from typing import Protocol


@dataclass
class SMSResult:
    success: bool
    provider_message_id: str | None = None
    error: str | None = None
    outcome_unknown: bool = False

class SMSProvider(Protocol):
    def send_sms(self, phone_number: str, message: str) -> SMSResult:
        ...


class DisabledSMSProvider:
    def send_sms(self, phone_number: str, message: str) -> SMSResult:
        return SMSResult(
            success=False,
            error="SMS sending is disabled. No SMS was sent.",
        )


def get_sms_provider():
    from app.core.config import settings

    if not settings.sms_live_enabled:
        return DisabledSMSProvider()

    if not (
        settings.vonage_api_key
        and settings.vonage_api_secret
        and settings.vonage_sender_id
    ):
        return DisabledSMSProvider()

    from app.services.vonage_sms_provider import VonageSMSProvider

    return VonageSMSProvider(
        api_key=settings.vonage_api_key,
        api_secret=settings.vonage_api_secret,
        sender_id=settings.vonage_sender_id,
    )

def get_emergency_notification_status(
    emergency_id: UUID,
    user_id: UUID,
):
    with SessionLocal() as db:

        emergency = db.scalar(
            select(Emergency).where(
                Emergency.emergency_id == str(emergency_id),
                Emergency.user_id == str(user_id),
            )
        )

        if emergency is None:
            raise PermissionError("Emergency record not found.")

        deliveries = db.scalars(
            select(NotificationDelivery)
            .where(
                NotificationDelivery.emergency_id
                == str(emergency_id)
            )
            .order_by(NotificationDelivery.created_at.asc())
        ).all()

        notifications = [
            NotificationDeliveryItem.model_validate(item)
            for item in deliveries
        ]

        messages = {
            "not_started": "Notification delivery has not started.",
            "ready": "Emergency approved and ready for queueing.",
            "queued": "Notifications are queued or awaiting retry.",
            "sent": "Notifications have been sent.",
            "partially_failed": "Some notifications failed.",
            "failed": "All notification delivery attempts failed.",
        }

        return EmergencyNotificationStatusResponse(
            emergency_id=emergency.emergency_id,
            delivery_status=emergency.delivery_status,
            recipient_count=emergency.recipient_count,
            notifications=notifications,
            message=messages.get(
                emergency.delivery_status,
                "Notification status retrieved.",
            ),
        )

@asynccontextmanager
async def lifespan(app: FastAPI):

    # Start background device monitoring
    monitor_task = asyncio.create_task(
        device_monitor_loop()
    )

    logger.info("SNATCH Device Monitor Started")

    # Start notification worker
    notification_stop_event = asyncio.Event()

    notification_task = asyncio.create_task(
        notification_worker_loop(
            notification_stop_event,
            poll_interval_seconds=15,
        )
    )

    logger.info("SNATCH Notification Worker Started")

    try:
        yield

    finally:
        # Stop notification worker gracefully
        notification_stop_event.set()

        try:
            await notification_task
        except asyncio.CancelledError:
            logger.info("SNATCH Notification Worker Stopped")

        # Stop device monitoring
        monitor_task.cancel()

        try:
            await monitor_task
        except asyncio.CancelledError:
            logger.info("SNATCH Device Monitor Stopped")

@app.post(
    "/api/v1/analyze",
    response_model=AnalysisResponse
)
async def analyze_device_event(
    event: DeviceEvent,
    current_user: User = Depends(get_current_user),
):
    # 1. Verify device ownership and active status.
    authenticated_user_id = UUID(str(current_user.user_id))

    device = get_device(
        device_id=event.device_id,
        user_id=authenticated_user_id,
    )

    if device is None or not device.is_active:
        raise HTTPException(
            status_code=404,
            detail="Active device not found",
        )

    # 2. Verify registered platform.
    if device.platform != event.platform:
        raise HTTPException(
            status_code=409,
            detail="Device platform mismatch",
        )

    # 3. Never trust user_id from request body.
    event_data = event.model_dump(mode="json")
    event_data["user_id"] = str(authenticated_user_id)

    try:
        initial_state = {
            "event": event_data,
            "agent_trace": [],
            "errors": [],
        }

        result = await snatch_graph.ainvoke(initial_state)

        assessment = result["risk_assessment"]

        return AnalysisResponse(
            event_id=event.event_id,
            risk_assessment=RiskAssessmentResponse(
                score=assessment["score"],
                level=assessment["level"],
                confidence=assessment["confidence"],
                signals=assessment["signals"],
                explanation=result["explanation"],
                recommended_action=assessment["recommended_action"],
            ),
            next_action=result["next_action"],
            agent_trace=result["agent_trace"],
            openai_used=result.get("openai_used", False),
        )

    except Exception as error:
        logger.exception("Legacy analysis workflow failed")

        raise HTTPException(
            status_code=500,
            detail="SNATCH workflow execution failed",
        ) from error

import asyncio
import logging

from app.services.device_service import mark_stale_devices_offline

logger = logging.getLogger(__name__)


async def device_monitor_loop():
    while True:
        try:
            count = await asyncio.to_thread(
                mark_stale_devices_offline,
                5
            )

            if count > 0:
                logger.info(
                    "Marked %s stale devices as offline",
                    count
                )

        except Exception:
            logger.exception("Device monitoring cycle failed")

        await asyncio.sleep(60)

def generate_test_signature(params, secret):
    signing_params = {
        key: value.replace("&", "_").replace("=", "_")
        for key, value in params.items()
        if key != "sig"
    }

    signing_string = "".join(
        f"&{key}={signing_params[key]}"
        for key in sorted(signing_params)
    )

    return hmac.new(
        secret.encode(),
        signing_string.encode(),
        hashlib.sha256,
    ).hexdigest()

    secret = "test-secret"
